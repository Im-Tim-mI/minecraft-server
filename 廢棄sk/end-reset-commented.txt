# ============================================
# 末地每週重置腳本
# 每週日中午 12:00 自動重置末地
# ============================================
# 檔案名稱：end-reset.sk
# 放置位置：plugins/Skript/scripts/
# ============================================
# 
# ⚠️ 重要設定：
# 請確保 Multiverse config.yml 中設定：
# confirm-mode: disable_console
#
# ============================================

# ============================================
# 變數設定區塊
# ============================================
# options 是 Skript 的常數定義區塊
# 可以在腳本中用 {@變數名} 來引用這些值
# 這樣做的好處是修改時只需要改這裡一處
# ============================================
options:
    # prefix: 所有廣播訊息的前綴
    # &6 = 金色, &l = 粗體, &r = 重置格式
    prefix: &6&l[末地重置] &r

# ============================================
# 定時檢查區塊
# ============================================
# every minute: 每分鐘執行一次這個區塊內的程式碼
# Skript 不支援 "every day at 11:30" 這種語法
# 所以我們用每分鐘檢查的方式來實現定時任務
# ============================================
every minute:
    # 取得當前時間，格式化為 "時:分" (例如: "11:30", "12:00")
    # HH = 24小時制的小時 (00-23)
    # mm = 分鐘 (00-59)
    set {_time} to now formatted as "HH:mm"
    
    # 取得當前是星期幾，格式化為英文全名
    # EEEE = 星期的完整名稱 (例如: "Sunday", "Monday")
    # {_變數名} 中的底線表示這是區域變數，只在這個區塊內有效
    set {_day} to now formatted as "EEEE"
    
    # 檢查今天是否為週日
    # 只有週日才執行以下的廣播和重置
    if {_day} is "Sunday":
        
        # ============================================
        # 11:30 - 第一次預告（提前 30 分鐘）
        # ============================================
        if {_time} is "11:30":
            # broadcast 會向伺服器所有玩家發送訊息
            # {@prefix} 會被替換成上面 options 中定義的前綴
            # &e = 黃色
            broadcast "{@prefix}&e12:00 末地將會重置！"
            broadcast "{@prefix}&e請末地的玩家提前離開並保存物品！"
        
        # ============================================
        # 11:50 - 第二次警告（提前 10 分鐘）
        # ============================================
        if {_time} is "11:50":
            # &c = 紅色，表示更緊急的警告
            broadcast "{@prefix}&c末地重置 倒數十分鐘！"
            broadcast "{@prefix}&c請立即離開末地！"
        
        # ============================================
        # 11:55 - 第三次警告（提前 5 分鐘）
        # ============================================
        if {_time} is "11:55":
            # &c&l = 紅色 + 粗體，強調緊急性
            broadcast "{@prefix}&c&l末地重置 倒數五分鐘！"
        
        # ============================================
        # 11:59 - 最後警告（提前 1 分鐘）
        # ============================================
        if {_time} is "11:59":
            # &4 = 深紅色，最高級別警告
            # &l = 粗體
            broadcast "{@prefix}&4&l末地重置 倒數一分鐘！"
            broadcast "{@prefix}&4&l所有末地玩家將被強制傳送！"
        
        # ============================================
        # 12:00 - 執行重置
        # ============================================
        if {_time} is "12:00":
            # execute console command 以控制台身份執行指令
            # 這會觸發下面定義的 /resetend 指令
            execute console command "resetend"

# ============================================
# 手動重置指令
# ============================================
# command /resetend: 定義一個新指令 /resetend
# 這個指令可以被管理員手動執行，也會被上面的定時任務呼叫
# ============================================
command /resetend:
    # permission: op 表示只有 OP（管理員）可以使用這個指令
    permission: op
    
    # description: 指令的描述，會顯示在 /help 中
    description: 手動重置末地
    
    # trigger: 當指令被執行時，執行以下的程式碼
    trigger:
        # 廣播重置開始訊息
        # &b = 水藍色
        broadcast "{@prefix}&b正在重置末地..."
        
        # ============================================
        # 步驟 1：傳送所有末地玩家到主世界
        # ============================================
        # loop all players in world "world_the_end":
        # 遍歷所有在 world_the_end 世界中的玩家
        # loop-player 是當前迴圈中的玩家
        loop all players in world "world_the_end":
            # teleport 將玩家傳送到指定位置
            # spawn of world "world" = 主世界的出生點
            teleport loop-player to spawn of world "world"
            
            # send 向特定玩家發送訊息
            # to loop-player 指定訊息接收者
            send "{@prefix}&e你已被傳送至主世界！" to loop-player
        
        # wait 2 seconds: 等待 2 秒
        # 確保所有玩家都已經被傳送完成
        # 避免在玩家還在末地時就刪除世界
        wait 2 seconds
        
        # ============================================
        # 步驟 2：匯入末地到 Multiverse
        # ============================================
        # 如果末地不在 Multiverse 管理中，先匯入它
        # 這一步確保末地被 Multiverse 正確識別
        # 如果已經匯入過，這個指令會顯示錯誤但不影響後續步驟
        execute console command "mv import world_the_end THE_END"
        
        # 等待 1 秒讓匯入完成
        wait 1 second
        
        # ============================================
        # 步驟 3：卸載末地
        # ============================================
        # mv unload: 從記憶體中卸載世界
        # --remove-players: 如果還有玩家在末地，強制傳送他們離開
        # 這是安全措施，確保沒有玩家會因為世界被刪除而出問題
        execute console command "mv unload world_the_end --remove-players"
        
        # 等待 1 秒讓卸載完成
        wait 1 second
        
        # ============================================
        # 步驟 4：刪除末地
        # ============================================
        # mv delete: 完全刪除世界（包括所有檔案）
        # 這會刪除末地的所有資料：地形、建築、末影龍狀態等
        # ⚠️ 這需要 Multiverse 配置 confirm-mode: disable_console
        #    否則會需要手動輸入確認碼
        execute console command "mv delete world_the_end"
        
        # 等待 2 秒讓刪除完成
        # 刪除需要更長時間，因為要處理檔案系統
        wait 2 seconds
        
        # ============================================
        # 步驟 5：重新建立末地
        # ============================================
        # mv create: 建立新世界
        # world_the_end: 世界名稱
        # THE_END: 世界類型（必須大寫）
        #   - NORMAL = 主世界
        #   - NETHER = 地獄
        #   - THE_END = 末地
        # 這會生成全新的末地，包括新的末影龍
        execute console command "mv create world_the_end THE_END"
        
        # 等待 3 秒讓世界生成完成
        # 世界生成需要較長時間
        wait 3 seconds
        
        # ============================================
        # 步驟 6：載入末地（已註解掉）
        # ============================================
        # mv create 會自動載入新建立的世界
        # 所以這一步其實不需要，已經註解掉
        # 如果取消註解，只會顯示 "World already loaded" 錯誤
        #execute console command "mv load world_the_end"
        
        # 等待 1 秒（保留以防需要）
        wait 1 second
        
        # ============================================
        # 廣播重置完成訊息
        # ============================================
        # 空白行讓訊息更明顯
        broadcast ""
        
        # &a = 綠色，表示成功
        # &l = 粗體
        broadcast "{@prefix}&a&l末地重置完成！"
        
        # &d = 粉紅色
        # ✦ 是裝飾用的特殊符號
        broadcast "{@prefix}&d&l✦ 搶龍大作戰開始！ ✦"
        
        # 結尾空白行
        broadcast ""

# ============================================
# 查看重置資訊指令
# ============================================
# 這個指令讓任何玩家都可以查看下次重置時間
# 沒有設定 permission，所以所有玩家都可以使用
# ============================================
command /endresetinfo:
    # 指令描述
    description: 查看末地重置資訊
    
    # 當指令被執行時
    trigger:
        # send 只會向執行指令的玩家發送訊息（不是 broadcast 全服）
        # &6 = 金色, &l = 粗體
        send "&6&l========== 末地重置資訊 =========="
        
        # &e = 黃色, &b = 水藍色
        send "&e下次重置時間: &b每週日 12:00"
        
        # &f = 白色
        send "&e重置內容: &f末地世界完全重置，末影龍重生"
        
        # 結尾分隔線
        send "&6&l=================================="


# ============================================
# Minecraft 顏色代碼參考
# ============================================
# &0 = 黑色      &1 = 深藍色    &2 = 深綠色    &3 = 深青色
# &4 = 深紅色    &5 = 紫色      &6 = 金色      &7 = 灰色
# &8 = 深灰色    &9 = 藍色      &a = 綠色      &b = 青色
# &c = 紅色      &d = 粉紅色    &e = 黃色      &f = 白色
#
# 格式代碼：
# &l = 粗體      &o = 斜體      &n = 底線      &m = 刪除線
# &k = 隨機字元  &r = 重置格式
# ============================================
